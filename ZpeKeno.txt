local Bypass = {}

local function hookNamecall(targetRemote, blockFire)
    local mt = getrawmetatable(targetRemote)
    if not mt then return end
    local old = mt.__namecall
    if not old then return end
    setreadonly(mt, false)
    mt.__namecall = newcclosure(function(self, ...)
        local method = getnamecallmethod()
        if self == targetRemote and blockFire and (method == "FireServer" or method == "InvokeServer") then
            return nil
        end
        return old(self, ...)
    end)
    setreadonly(mt, true)
end

function Bypass.BlockTelemetry()
    local blocked = {
        "SendTelemetry", "LogAction", "AntiCheatReport",
        "ReportPlayer", "CheckSpeed", "CheckPosition",
        "VerifyCharacter", "HeartbeatCheck",
    }
    for _, remote in pairs(game:GetDescendants()) do
        if remote:IsA("RemoteEvent") or remote:IsA("RemoteFunction") then
            for _, b in pairs(blocked) do
                if remote.Name:lower():find(b:lower()) then
                    pcall(hookNamecall, remote, true)
                end
            end
        end
    end
    print("[Zpekeno] Telemetria bloqueada.")
end

function Bypass.SpeedSpoof(realSpeed, fakeSpeed)
    realSpeed = realSpeed or 16
    fakeSpeed = fakeSpeed or 16
    local plr = game.Players.LocalPlayer
    local char = plr.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end

    local mt = getrawmetatable(game)
    if not mt then return end
    local oldIndex = mt.__index
    local oldNewIndex = mt.__newindex
    setreadonly(mt, false)

    mt.__index = newcclosure(function(self, key)
        if self == hum and key == "WalkSpeed" then return fakeSpeed end
        return oldIndex(self, key)
    end)
    mt.__newindex = newcclosure(function(self, key, value)
        if self == hum and key == "WalkSpeed" then
            oldNewIndex(self, key, realSpeed)
            return
        end
        return oldNewIndex(self, key, value)
    end)
    setreadonly(mt, true)

    hum.WalkSpeed = realSpeed
    print("[Zpekeno] Speed spoof:", realSpeed, "->", fakeSpeed)
end

function Bypass.SafeTeleport(targetCFrame, steps, duration)
    steps = steps or 10
    duration = duration or 0.15
    local plr = game.Players.LocalPlayer
    local char = plr.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    local start = root.CFrame
    for i = 1, steps do
        local a = i / steps
        root.CFrame = start:Lerp(targetCFrame, a)
        task.wait(duration / steps)
    end
end

function Bypass.SafeDash(direction, speed)
    local plr = game.Players.LocalPlayer
    local char = plr.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    local bv = Instance.new("BodyVelocity")
    bv.MaxForce = Vector3.new(1e5, 0, 1e5)
    bv.Velocity = direction * speed
    bv.Name = "Velocity"
    bv.Parent = root
    task.wait(0.15)
    if bv and bv.Parent then bv:Destroy() end
end

function Bypass.CleanClientChecks()
    pcall(function()
        local env = getfenv(0)
        local keys = {}
        for k in pairs(env) do
            local s = tostring(k):lower()
            if s:find("anticheat") or s:find("detection") or s:find("bfcheck") then
                table.insert(keys, k)
            end
        end
        for _, k in pairs(keys) do env[k] = nil end
    end)
    print("[Zpekeno] Client checks limpos.")
end

function Bypass.ProtectExecutor()
    pcall(function()
        local env = getfenv(0)
        env.getgenv = nil
        env.getrawmetatable = nil
        env.setreadonly = nil
        env.newcclosure = nil
    end)
end

function Bypass.Init(settings)
    settings = settings or {}
    Bypass.BlockTelemetry()
    Bypass.CleanClientChecks()
    Bypass.ProtectExecutor()
    Bypass.SpeedSpoof(settings.RealSpeed or 50, settings.FakeSpeed or 16)
    print("[Zpekeno] Bypass carregado.")
end

    local char = target.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    local billboard = Instance.new("BillboardGui")
    billboard.Name = "Zpekeno_ESP"
    billboard.Size = UDim2.new(0, 200, 0, 100)
    billboard.StudsOffset = Vector3.new(0, 3, 0)
    billboard.AlwaysOnTop = true
    billboard.Adornee = root
    billboard.Parent = root

    if Config.Settings.Visual.PlayerOutline then
        local hl = Instance.new("Highlight")
        hl.Name = "SelectionBox"
        hl.FillColor = Config.Settings.Visual.OutlineColor
        hl.OutlineColor = Config.Settings.Visual.OutlineColor
        hl.FillTransparency = 0.7
        hl.OutlineTransparency = 0
        hl.Parent = char
    end

    if Config.Settings.Visual.HealthPercent then
        local lbl = Instance.new("TextLabel")
        lbl.Name = "Health"
        lbl.Size = UDim2.new(1, 0, 0.3, 0)
        lbl.Position = UDim2.new(0, 0, 0, 0)

        lbl.TextStrokeTransparency = 0
        lbl.Font = Enum.Font.GothamBold
        lbl.TextScaled = true
        lbl.Text = "100%"
        lbl.Parent = billboard
    end

    if Config.Settings.Visual.Distance then
        local lbl = Instance.new("TextLabel")
        lbl.Name = "Distance"
        lbl.Size = UDim2.new(1, 0, 0.3, 0)
        lbl.Position = UDim2.new(0, 0, 0.3, 0)
        lbl.BackgroundTransparency = 1
        lbl.TextColor3 = Color3.fromRGB(255, 255, 0)
        lbl.TextStrokeTransparency = 0
        lbl.Font = Enum.Font.GothamBold
        lbl.TextScaled = true
        lbl.Text = "0m"
        lbl.Parent = billboard
    end

    drawings[target] = billboard
end

function Visual.RemoveESP(target)
    if drawings[target] then
        drawings[target]:Destroy()
        drawings[target] = nil
    end
    local char = target.Character
    if char then
        local h = char:FindFirstChild("SelectionBox")
        if h then h:Destroy() end
    end
end

function Visual.Update()
    local plr = game.Players.LocalPlayer
    local char = plr.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    for target, gui in pairs(drawings) do
        local tchar = target.Character
        if not tchar then
            Visual.RemoveESP(target)
        else
            local troot = tchar:FindFirstChild("HumanoidRootPart")
            local thum = tchar:FindFirstChildOfClass("Humanoid")
            if not troot or not thum then
                Visual.RemoveESP(target)
            else
                local dist = (troot.Position - root.Position).Magnitude
                if Config.Settings.Visual.HealthPercent then
                    local hp = gui:FindFirstChild("Health")
                    if hp then hp.Text = math.floor((thum.Health / thum.MaxHealth) * 100) .. "%" end
                end
                if Config.Settings.Visual.Distance then
                    local dl = gui:FindFirstChild("Distance")
                    if dl then dl.Text = math.floor(dist) .. "m" end
                end
            end
        end
    end
end

function Visual.StartLoop()
    task.spawn(function()
        while task.wait(0.1) do Visual.Update() end
    end)
end

function Visual.RefreshAll()
    for _, plr in pairs(game.Players:GetPlayers()) do
        if plr ~= game.Players.LocalPlayer then Visual.CreateESP(plr) end
    end
end

local Aimbot = {}
local currentTarget = nil

function Aimbot.GetClosestTarget()
    local plr = game.Players.LocalPlayer
    local char = plr.Character
    if not char then return nil end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return nil end

    local closest, minDist = nil, Config.Settings.Aimbot.TargetDistance.max
    for _, target in pairs(game.Players:GetPlayers()) do
        if target == plr then continue end
        if Config.Settings.Aimbot.TeamFilter and target.Team == plr.Team then continue end
        local tc = target.Character
        if tc then
            local tr = tc:FindFirstChild("HumanoidRootPart")
            local th = tc:FindFirstChildOfClass("Humanoid")
            if tr and th and th.Health > 0 then
                local d = (tr.Position - root.Position).Magnitude
                if d >= Config.Settings.Aimbot.TargetDistance.min and d < minDist then
                    minDist = d
                    closest = target
                end
            end
        end
    end
    return closest
end

function Aimbot.PredictPosition(target)
    local plr = game.Players.LocalPlayer
    local char = plr.Character
    if not char then return nil end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return nil end
    local tc = target.Character
    if not tc then return nil end
    local tr = tc:FindFirstChild("HumanoidRootPart")
    if not tr then return nil end

    local dist = (tr.Position - root.Position).Magnitude
    local speed = 100
    local pred = Config.Settings.Aimbot.Prediction / 100
    local travel = dist / speed
    return tr.Position + (tr.AssemblyLinearVelocity * travel * pred)
end

function Aimbot.Start()
    task.spawn(function()
        while task.wait(0.05) do
            if Config.Settings.Aimbot.TargetAssist then
                local target = Aimbot.GetClosestTarget()
                if target then
                    currentTarget = target
                    local predPos = Aimbot.PredictPosition(target)
                    if predPos then
                        local cam = workspace.CurrentCamera
                        if cam then
                            local goal = CFrame.new(cam.CFrame.Position, predPos)
                            cam.CFrame = cam.CFrame:Lerp(goal, Config.Settings.Aimbot.Smoothness)
                        end
                    end
                end
            end
        end
    end)
end

function Aimbot.GetTarget() return currentTarget end

local Other = {}

function Other.SetSpeed(speed)
    local plr = game.Players.LocalPlayer
    local char = plr.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum then hum.WalkSpeed = speed end
end

function Other.SetJump(jump)
    local plr = game.Players.LocalPlayer
    local char = plr.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum then hum.JumpPower = jump; hum.UseJumpPower = true end
end

function Other.Dash()
    if not Config.Settings.Other.DashEnabled then return end
    local plr = game.Players.LocalPlayer
    local char = plr.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not root or not hum then return end
    local dir = hum.MoveDirection
    if dir.Magnitude < 0.1 then dir = root.CFrame.LookVector end
    Bypass.SafeDash(dir, Config.Settings.Other.Dash)
end

function Other.TeleportTo(cframe)
    Bypass.SafeTeleport(cframe, Config.Settings.Bypass.TeleportSteps, Config.Settings.Bypass.TeleportDuration)
end

function Other.StartAutoApply()
    task.spawn(function()
        while task.wait(0.5) do
            Other.SetSpeed(Config.Settings.Other.MovementSpeed)
            Other.SetJump(Config.Settings.Other.JumpHeight)
        end
    end)
end
