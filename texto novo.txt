    sidebar   = Color3.fromRGB(12, 15, 22),

local UIS = game:GetService("UserInputService")

local function makeFrame(props)
    local f = Instance.new("Frame")
    f.Active = true
    f.BorderSizePixel = 0
    f.BackgroundColor3 = COLORS.panel
    for k, v in pairs(props) do f[k] = v end
    return f
end

local function makeButton(props)
    local b = Instance.new("TextButton")
    b.Active = true
    b.AutoButtonColor = false
    b.BorderSizePixel = 0
    b.BackgroundColor3 = COLORS.panel
    b.Text = ""
    b.Font = Enum.Font.Gotham
    b.TextColor3 = COLORS.text
    for k, v in pairs(props) do b[k] = v end
    return b
end

local function makeLabel(props)
    local l = Instance.new("TextLabel")
    l.BackgroundTransparency = 1
    l.Font = Enum.Font.Gotham
    l.TextColor3 = COLORS.text
    l.TextXAlignment = Enum.TextXAlignment.Left
    l.ZIndex = 2
    for k, v in pairs(props) do l[k] = v end
    return l
end

local function round(obj, r)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, r or 8)
    c.Parent = obj
    return c
end

local function stroke(obj, color, thickness, transparency)
    local s = Instance.new("UIStroke")
    s.Color = color or COLORS.border
    s.Thickness = thickness or 1
    s.Transparency = transparency or 0
    s.Parent = obj
    return s
end

        local cam = workspace.CurrentCamera
        local vp = cam and cam.ViewportSize or Vector2.new(1920, 1080)
        local isMobile = UIS.TouchEnabled and not UIS.KeyboardEnabled
        local isTablet = UIS.TouchEnabled and UIS.KeyboardEnabled
        local isPortrait = vp.Y > vp.X
        if isMobile then
            return isPortrait and 0.55 or 0.65
        elseif isTablet then
            return 0.85
        else
            if vp.X < 1366 then return 0.85 end
            if vp.X < 1920 then return 1.0 end

    local function addToggleRow(parent, labelText, subText, state, callback)
        local count = 0
        for _ in pairs(parent:GetChildren()) do count = count + 1 end

        local row = makeButton({
            Size = UDim2.new(1, 0, 0, 44),

        })

            Text = labelText,
            Font = Enum.Font.GothamBold,
            TextSize = 12,
            TextColor3 = COLORS.text,
            Size = UDim2.new(1, -70, 0, 16),
            Position = UDim2.new(0, 10, 0, 6),
            ZIndex = 17,
            Parent = row,

            makeLabel({
                Text = subText,
                Font = Enum.Font.Gotham,
                TextSize = 9,
                TextColor3 = COLORS.subtext,
                Size = UDim2.new(1, -70, 0, 12),
                Position = UDim2.new(0, 10, 0, 22),
                ZIndex = 17,
                Parent = row,
            })
        end

        local sw = makeFrame({
            Size = UDim2.new(0, 40, 0, 20),
            Position = UDim2.new(1, -50, 0.5, -10),
            BackgroundColor3 = state and COLORS.accent or COLORS.off,
            Parent = row,
            ZIndex = 17,
        })
        round(sw, 10)

        local knob = makeFrame({
            Size = UDim2.new(0, 16, 0, 16),
            Position = state and UDim2.new(1, -18, 0, 2) or UDim2.new(0, 2, 0, 2),
            BackgroundColor3 = Color3.fromRGB(255, 255, 255),
            Parent = sw,
            ZIndex = 18,
        })
        round(knob, 8)

        row.MouseButton1Click:Connect(function()
            state = not state
            callback(state)
            sw.BackgroundColor3 = state and COLORS.accent or COLORS.off
            knob.Position = state and UDim2.new(1, -18, 0, 2) or UDim2.new(0, 2, 0, 2)
        end)
    end

        for _ in pairs(parent:GetChildren()) do count = count + 1 end

        local row = makeFrame({
            Size = UDim2.new(1, 0, 0, 62),

            Position = UDim2.new(0, 10, 0, 6),

                TextColor3 = COLORS.subtext,

                Parent = row,
            })

        local badge = makeFrame({
            Size = UDim2.new(0, 80, 0, 18),
            Position = UDim2.new(1, -90, 0, 8),
            BackgroundColor3 = COLORS.panel,
            Parent = row,
            ZIndex = 17,

        round(badge, 4)
        stroke(badge, COLORS.border, 1, 0.5)

            Text = defaultVal .. " - " .. maxVal,
            Font = Enum.Font.GothamBold,
            TextSize = 10,
            TextColor3 = COLORS.text,
            Size = UDim2.new(1, 0, 1, 0),
            TextXAlignment = Enum.TextXAlignment.Center,
            TextYAlignment = Enum.TextYAlignment.Center,
            ZIndex = 18,
            Parent = badge,

        local track = makeFrame({
            Size = UDim2.new(1, -20, 0, 6),
            Position = UDim2.new(0, 10, 0, 42),
            BackgroundColor3 = COLORS.off,
            Parent = row,
            ZIndex = 17,
        })
        round(track, 3)

        local fill = makeFrame({
            Size = UDim2.new((defaultVal - minVal) / (maxVal - minVal), 0, 1, 0),
            BackgroundColor3 = COLORS.accent,
            Parent = track,
            ZIndex = 18,
        })
        round(fill, 3)

        local thumb = makeFrame({
            Size = UDim2.new(0, 14, 0, 14),

            Position = UDim2.new(1, -40, 0, 50),
            TextXAlignment = Enum.TextXAlignment.Right,
            ZIndex = 17,
            Parent = row,
        })

        local dragging = false
        local function updateSlider(input)
            local trackAbs = track.AbsolutePosition
            local trackSize = track.AbsoluteSize
            local rel = math.clamp((input.Position.X - trackAbs.X) / trackSize.X, 0, 1)
            local val = math.floor(minVal + (maxVal - minVal) * rel)
            fill.Size = UDim2.new(rel, 0, 1, 0)
            thumb.Position = UDim2.new(rel, -7, 0.5, -7)
            badgeLbl.Text = val .. " - " .. maxVal
            callback(val)
        end

        track.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1
               or input.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                updateSlider(input)
            end
        end)
        UIS.InputChanged:Connect(function(input)
            if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
               or input.UserInputType == Enum.UserInputType.Touch) then
                updateSlider(input)
            end
        end)
        UIS.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1
               or input.UserInputType == Enum.UserInputType.Touch then
                dragging = false
            end
        end)
    end

    otherCol.Parent.Position = UDim2.new(0, 0, 0, 0)

    addToggleRow(otherCol, "Dash Enabled", "Toggle dash on/off.", Config.Settings.Other.DashEnabled, function(v)
        Config.Settings.Other.DashEnabled = v
    end)
    addSliderRow(otherCol, "Dash", "Increase dash distance/speed.", 5, 50, Config.Settings.Other.Dash, function(v)
        Config.Settings.Other.Dash = v
    end)
    addSliderRow(otherCol, "Movement Speed", "Adjust player movement speed.", 5, 100, Config.Settings.Other.MovementSpeed, function(v)
        Config.Settings.Other.MovementSpeed = v
    end)
    addSliderRow(otherCol, "Jump Height", "Increase jump height.", 5, 50, Config.Settings.Other.JumpHeight, function(v)
        Config.Settings.Other.JumpHeight = v
    end)

    --==========================================================
    -- DRAG (funciona em PC e mobile)
    --==========================================================
    local dragging, dragStart, startPos
    local function beginDrag(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
           or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = main.Position
        end
    end
    local function updateDrag(input)
        if not dragging then return end
        if input.UserInputType ~= Enum.UserInputType.MouseMovement
           and input.UserInputType ~= Enum.UserInputType.Touch then return end
        local delta = input.Position - dragStart
        main.Position = UDim2.new(
            startPos.X.Scale, startPos.X.Offset + delta.X,
            startPos.Y.Scale, startPos.Y.Offset + delta.Y
        )
    end
    local function endDrag(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
           or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end

    header.InputBegan:Connect(beginDrag)
    header.InputChanged:Connect(updateDrag)
    header.InputEnded:Connect(endDrag)
    UIS.InputChanged:Connect(updateDrag)
    UIS.InputEnded:Connect(endDrag)

    -- centraliza
    local cam2 = workspace.CurrentCamera
    if cam2 then
        local vp = cam2.ViewportSize
        local scale = getScale()
        local w = 800 * scale
        local h = 500 * scale
        main.Position = UDim2.new(0, math.max(10, (vp.X - w) / 2), 0, math.max(10, (vp.Y - h) / 2))
    end

    print("[Zpekeno] UI carregada.")
    return screen
